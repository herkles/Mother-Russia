je_mat_rossiya_great_reforms_poland = {
    icon = "gfx/interface/icons/law_icons/neo_absolutism.dds"

	group = je_group_russia_great_reformer

    immediate = {
		c:RUS = {# Got to have Alexander II
			ruler = { save_scope_as = alexander_ii_scope }
		}
    }

	scripted_button = mat_rossiya_poland_end_martial_law
	scripted_button = mat_rossiya_restore_polish_rights

    complete = {
		# Got to remove martial law
		NOT = {
			active_law:lawgroup_distribution_of_power ?= {
				has_amendment = amendment_type:mat_rossiya_amendment_polish_martial_law
			}
		}
		# got to restore the polish rights and Church
        country_has_primary_culture = cu:polish
		has_law_or_variant = law_type:law_freedom_of_conscience
		# serfdom needs to be abolished
        NOT = {
            has_law_or_variant = law_type:law_serfdom
        }
		# expand the university of Warsaw
		any_scope_building = {
			is_building_type = building_university
			level >= 2
			occupancy > 0.9
		}
		# expand the beaucracy
		any_scope_building = {
			is_building_type = building_government_administration
			NOT = { has_active_production_method = pm_simple_organization }
			level >= 2
			occupancy >= 0.9
		}
		bureaucracy >= 0
		approaching_bureaucracy_shortage = no
		# be loyal to the tsar - low liberty desire
        liberty_desire < 10
		radical_fraction = {
			value < 0.20
		}
    }

    on_complete = {
		# Restoration of the Constitution of Poland if you decide to go down this path
		show_as_tooltip = {
			add_loyalists = {
                value = medium_radicals
				culture = cu:polish
            }
			create_diplomatic_pact = {
				country = c:RUS
				type = personal_union
			}
		}
		trigger_event = { id = mat_rossiya_poland.051 popup = yes }
    }

    fail = {
		# do not have a revolution or seek to leave poland
        liberty_desire >= 80
		has_revolution = yes
        custom_tooltip = {# Got to have Alexander II
            text = ruler_not_alexander_ii_tt
			c:RUS = {
				ruler = {
                	NOT = { has_template = RUS_alexander_ii }
            	}
			}
        }
    }

	on_fail = {
		# The January Uprising happens
		custom_tooltip = mat_rossiya_january_uprising_tt
		trigger_event = { id = mat_rossiya_poland.050 popup = yes }

	}

    weight = 1000
    should_be_pinned_by_default = yes
}
