


# Revolts and Uprisings
mat_rossiya_january_uprising_effects = {# january uprising
	c:POL = {# set politics for the January Uprising
		set_state_religion = rel:catholic
		remove_primary_culture = cu:russian
		add_primary_culture = cu:polish
		set_variable = mat_rossiya_poland_january_uprising
		# Power Structure Laws
		activate_law = law_type:law_parliamentary_republic
		activate_law = law_type:law_oligarchy
		activate_law = law_type:law_racial_segregation
		activate_law = law_type:law_freedom_of_conscience
		activate_law = law_type:law_mass_conscription
		activate_law = law_type:law_national_guard
		# Economic Laws
		activate_law = law_type:law_interventionism
		activate_law = law_type:law_homesteading
	

		ig:ig_armed_forces = {
			remove_ideology = ideology_enclavist
			add_ideology = ideology_patriotic
			add_ruling_interest_group = yes
		}
		ig:ig_devout = {
			remove_ideology = mat_rossiya_ideology_catholic_moralism 
			add_ideology = ideology_moralist
		}
		ig:ig_petty_bourgeoisie = {
			add_ruling_interest_group = yes
		}
		ig:ig_intelligentsia = {
			add_ruling_interest_group = yes
		}
	}
	c:RUS ?= {
		create_diplomatic_play = {
			name = january_uprising
			type = dp_january_uprising
			target_state = c:POL.capital

			add_war_goal = {
				holder = c:POL
				type = independence
				target_country = c:RUS
				primary_demand = yes
			}
		}
	}
	random_diplomatic_play = {# Extent of the rebellion; so if they win they get their claims
		limit = {
			is_diplomatic_play_type = dp_january_uprising
			target = {
				c:POL ?= this
			}
		}
		if = {
			limit = { exists = s:STATE_KAUNAS.region_state:RUS }
			add_war_goal = {
				holder = c:POL
				type = conquer_state
				target_state = s:STATE_KAUNAS.region_state:RUS
			}
		}
		if = {
			limit = { exists = s:STATE_VILNIUS.region_state:RUS }
			add_war_goal = {
				holder = c:POL
				type = conquer_state
				target_state = s:STATE_VILNIUS.region_state:RUS
			}
		}
		if = {
			limit = { exists = s:STATE_BREST.region_state:RUS }
			add_war_goal = {
				holder = c:POL
				type = conquer_state
				target_state = s:STATE_BREST.region_state:RUS
			}
		}
		if = {
			limit = { exists = s:STATE_VOLHYNIA.region_state:RUS }
			add_war_goal = {
				holder = c:POL
				type = conquer_state
				target_state = s:STATE_VOLHYNIA.region_state:RUS
			}
		}
	}
	s:STATE_MAZOVIA = {# Liberation
		set_owner_of_provinces = {# Congress Poland
			country = c:POL
			provinces = {
				x4969E4
				xC06121
				x0087C9
				x117AE1
				xDE551D        
			}
		}	
	}
	c:POL = {
		create_military_formation = {# Give them some men to fight back
			type = army
			hq_region = sr:region_poland
			combat_unit = {
				type = unit_type:combat_unit_type_irregular_infantry
				state_region = s:STATE_LESSER_POLAND
				count = 7
			}
			combat_unit = {
				type = unit_type:combat_unit_type_hussars
				state_region = s:STATE_LESSER_POLAND
				count = 5
			}
		}
	}
	c:POL = {
		random_interest_group = {
			limit = {
				leader = { has_culture = cu:polish }
			}
			leader = {
				set_character_as_ruler = yes
			}
		}
	}
	c:POL = {
		every_scope_character = {
			limit = {
				has_culture = cu:russian
			}
			transfer_character = c:RUS
		}
	}


	
}
