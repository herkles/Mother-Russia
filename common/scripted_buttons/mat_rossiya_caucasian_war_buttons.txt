### Russian Empire Buttons
mat_rossiya_russia_escalate_caucasian_war = {
	name = "russia_escalate_caucasian_war"
	desc = "russia_escalate_caucasian_war_desc"
	visible = {
		c:RUS ?= this # Got to be Russia
		is_subject = no
		NOT = {# can't have already conquered the regions
			owns_entire_state_region = STATE_KUBAN
			owns_entire_state_region = STATE_NORTH_CAUCASUS
			owns_entire_state_region = STATE_DAGESTAN
		}
		AND = {
			any_scope_state = {
				state_region = s:STATE_KUBAN
			}
			any_scope_state = {
				state_region = s:STATE_NORTH_CAUCASUS
			}
			any_scope_state = {
				state_region = s:STATE_DAGESTAN
			}
		}
		OR = {
			AND = {
				exists = c:CHC
				has_truce_with = c:CHC
			}
			AND = {
				exists = c:CIR
				has_truce_with = c:CIR
			}
		}
        NOT = {# Can't escalate more then once
        	has_variable = escalated_caucasian_war
        }
	}

	possible = {
		c:RUS ?= {# Russia must be a great power
			country_rank >= rank_value:great_power
		}
		OR = {
			s:STATE_KUBAN = {
				any_scope_state = {
					OR = {# Kuban is owned by either Circassia or Imamate
						AND = {
							exists = c:CIR
							owner = c:CIR
						}
						AND = {
							exists = c:CHC
							owner = c:CHC
						}
					}
				}
			}
			s:STATE_DAGESTAN = {
				any_scope_state = {
					OR = {
						AND = {
							exists = c:CIR
							owner = c:CIR
						}
						AND = {
							exists = c:CHC
							owner = c:CHC
						}
					}
				}
			}
			s:STATE_NORTH_CAUCASUS = {
				any_scope_state = {
					OR = {
						AND = {
							exists = c:CIR
							owner = c:CIR
						}
						AND = {
							exists = c:CHC
							owner = c:CHC
						}
					}
				}
			}
		}
		has_technology_researched = rifling
		has_technology_researched = logistics
	}

	effect = {
		set_variable = escalated_caucasian_war
		end_truce = c:CHC
		end_truce = c:CIR
	}

	ai_chance = {
		value = 15
	}
}

### Imamate Buttons
mat_rossiya_imamate_dispatch_muhammad_amin = {
	name = "imamate_dispatch_muhammad_amin"
	desc = "imamate_dispatch_muhammad_amin_desc"
	visible = {
		c:CHC ?= this # Got to be the Imamate
		exists = c:CIR
		is_subject = no
		c:CIR ?= {# Check to ensure they are involved with the Caucasian War
			is_involved_in_journal_entry = je_mat_rossiya_caucasian_war
		}
		NOT = {# can't have sent Muhammad Amin already
			has_variable = sent_muhammad_amin_var
		}
	}

	possible = {
		is_subject = no
		c:CIR ?= {# Check to ensure they are involved with the Caucasian War
			is_involved_in_journal_entry = je_mat_rossiya_caucasian_war
		}
		custom_tooltip = {
			text = muhammad_amin_possible
			c:CIR ?= {
				ruler = {
					NOT = {
						OR = {
							has_variable = ismail_var
							has_variable = hawduqo_var
						}
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = circassia_muhammad_amin_tt
		}
		c:CIR ?= {
			trigger_event = { id = mat_rossiya_caucasian_war.030 }
		}
		set_variable = sent_muhammad_amin_var
	}

	ai_chance = {
		value = 15
		if = {
			limit = {
				c:CIR.relations:root < relations_threshold:neutral
			}
			add = -30
		}
	}
}
### Subject Buttons
mat_rossiya_caucasian_disloyalty_button = {
	name = "mat_rossiya_caucasian_loyalty_button"
	desc = "mat_rossiya_caucasian_loyalty_button_desc"

	visible = {
		is_player = yes
		is_subject_of = c:RUS
		capital ?= { region = sr:region_caucasus }
	}
	selected = {
		has_diplomatic_pact = {
			who = c:RUS
			type = mat_rossiya_disloyal_subject
		}
	}
	possible = {
		trigger_if = {
			limit = {
				NOT = {
					has_diplomatic_pact = {
						who = c:RUS
						type = mat_rossiya_disloyal_subject
					}
				}
			}
			c:RUS.relations:root <= relations_threshold:poor
		}
		custom_tooltip = {
			text = mat_rossiya_caucasian_loyalty_button_cooldown_trigger
			NOT = {
				has_variable = mat_rossiya_caucasian_button_cooldown_var
			}
		}
	}
	effect = {
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:RUS
					type = mat_rossiya_disloyal_subject
				}
			}
			remove_diplomatic_pact = {
				country = c:RUS
				type = mat_rossiya_disloyal_subject
			}
		}
		else = {
			create_diplomatic_pact = {
				country = c:RUS
				type = mat_rossiya_disloyal_subject
			}
			custom_tooltip = {
				text = mat_rossiya_caucasian_loyalty_button_effect
				set_variable = {
					name = mat_rossiya_caucasian_button_cooldown_var
					years = 2
				}
			}
		}
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:RUS
					type = mat_rossiya_loyal_subject
				}
			}
			remove_diplomatic_pact = {
				country = c:RUS
				type = mat_rossiya_loyal_subject
			}
		}
	}
	ai_chance = {
		value = -1000
	}
}
